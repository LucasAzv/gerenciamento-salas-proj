CREATE TABLE usuarios (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    login VARCHAR(80) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL,
    role VARCHAR(20) NOT NULL,

    CONSTRAINT chk_role
        CHECK (role IN ('ADMIN', 'PROFESSOR'))
);


CREATE TABLE professores (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    cpf VARCHAR(11) NOT NULL UNIQUE,
    disciplina VARCHAR(255) NOT NULL,
    tipo_aula VARCHAR(20) NOT NULL,
    usuario_id BIGINT NOT NULL UNIQUE,

    CONSTRAINT fk_professor_usuario
        FOREIGN KEY (usuario_id) REFERENCES usuarios(id),

    CONSTRAINT chk_tipo_aula
        CHECK (tipo_aula IN ('PRATICA', 'TEORICA'))
);


CREATE TABLE salas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    tamanho INTEGER NOT NULL,
    tipo_sala VARCHAR(20) NOT NULL,
    disponivel BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT chk_tamanho
        CHECK (tamanho > 0),

    CONSTRAINT chk_tipo_sala
        CHECK (tipo_sala IN ('PRATICA', 'TEORICA'))
);


CREATE TABLE requisicoes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    professor_id BIGINT NOT NULL,
    sala_id BIGINT NOT NULL,
    data DATE NOT NULL,
    hora_inicio TIME(6) NOT NULL,
    hora_fim TIME(6) NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'PENDENTE',

    CONSTRAINT fk_requisicao_professor
        FOREIGN KEY (professor_id) REFERENCES professores(id),

    CONSTRAINT fk_requisicao_sala
        FOREIGN KEY (sala_id) REFERENCES salas(id),

    CONSTRAINT chk_status
        CHECK (status IN (
            'PENDENTE',
            'APROVADA',
            'REJEITADA',
            'CANCELADA'
        )),

    CONSTRAINT chk_horario
        CHECK (hora_inicio < hora_fim)
);


CREATE INDEX idx_reserva_sala_data
    ON requisicoes (sala_id, data, status);
